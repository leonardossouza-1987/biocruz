Tres trechos comentados

1. Validacao de CPF com digitos verificadores

// Valida CPF testando os dois digitos
const cpfValido = cpf => {
  cpf = cpf.replace(/\D/g, '');
  if (cpf.length !== 11 ||
      /^(\d)\1+$/.test(cpf)) return false;
  let s = 0;
  for (let i = 0; i < 9; i++)
    s += parseInt(cpf[i]) * (10 - i);
  let d1 = 11 - (s % 11);
  if (d1 >= 10) d1 = 0;
  if (d1 !== parseInt(cpf[9])) return false;
  // ... segundo digito
  return true;
};

2. IntersectionObserver do contador

// Anima o numero apenas quando entra na tela
const io = new IntersectionObserver(entries => {
  entries.forEach(entry => {
    if (!entry.isIntersecting) return;
    const alvo = +el.dataset.target;
    const inicio = performance.now();
    const animar = agora => {
      const p = Math.min((agora - inicio) / 1800, 1);
      // easing easeOutExpo
      const eased = p === 1 ? 1 : 1 - Math.pow(2, -10 * p);
      el.textContent = Math.floor(eased * alvo);
      if (p < 1) requestAnimationFrame(animar);
    };
    requestAnimationFrame(animar);
  });
}, { threshold: .5 });
